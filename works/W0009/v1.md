# Domino tilings of a 3 by 12 rectangle

How many different ways can a rectangle 3 squares tall and 12 squares wide be completely covered by 1 by 2 dominoes (each domino may be horizontal or vertical, no overlaps, no gaps)?

Answer with a single whole number.

```verify
def verify(answer):
    # Count tilings of a 3 x n board with a transfer over column profiles (3 bits each).
    def tilings(n):
        def fill(row, cur, nxt, out):
            if row == 3:
                out.append(nxt)
                return
            if cur >> row & 1:
                fill(row + 1, cur, nxt, out)
                return
            fill(row + 1, cur, nxt | 1 << row, out)          # horizontal domino into the next column
            if row < 2 and not cur >> (row + 1) & 1:
                fill(row + 2, cur, nxt, out)                 # vertical domino in this column
        ways = {0: 1}
        for _ in range(n):
            new = {}
            for state, count in ways.items():
                nexts = []
                fill(0, state, 0, nexts)
                for s in nexts:
                    new[s] = new.get(s, 0) + count
            ways = new
        return ways.get(0, 0)
    return answer.strip() == str(tilings(12))
```
